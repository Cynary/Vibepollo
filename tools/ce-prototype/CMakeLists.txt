cmake_minimum_required(VERSION 3.20)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(direct_capture_launcher LANGUAGES CXX)
endif()

if(NOT WIN32)
    message(FATAL_ERROR "The direct capture launcher requires Windows")
endif()

add_executable(ce-stream-launcher WIN32 stream_launcher.cpp)
target_compile_features(ce-stream-launcher PRIVATE cxx_std_20)
if(MINGW)
    target_link_options(ce-stream-launcher PRIVATE -municode -static)
endif()

# Explicit developer-only targets; no observer is installed with the host.
option(CE_BUILD_TIMING_TOOLS "Build the frame identity and timestamp observer tools" OFF)
if(CE_BUILD_TIMING_TOOLS)
    set(CE_SOURCE_DIR "" CACHE PATH "Matching CaptureEngine source tree")
    if(NOT EXISTS "${CE_SOURCE_DIR}/hook/common/present_observer.h")
        message(FATAL_ERROR "CE_SOURCE_DIR must contain the matching observer header")
    endif()
    add_executable(ce-frame-probe WIN32 timing/frame_probe.cpp)
    add_executable(ce-timestamp-observer timing/timestamp_observer.cpp)
    add_executable(ce-etl-frame-window timing/etl_frame_window.cpp)
    foreach(target ce-frame-probe ce-timestamp-observer ce-etl-frame-window)
        target_compile_features(${target} PRIVATE cxx_std_23)
        if(MINGW)
            target_link_options(${target} PRIVATE -static)
        endif()
    endforeach()
    if(MINGW)
        target_link_options(ce-frame-probe PRIVATE -municode)
        target_link_options(ce-timestamp-observer PRIVATE -municode)
    endif()
    target_include_directories(ce-timestamp-observer PRIVATE "${CE_SOURCE_DIR}")
    target_link_libraries(ce-frame-probe PRIVATE d3d11 dxgi ole32 oleaut32 uuid runtimeobject dwmapi)
    target_link_libraries(ce-timestamp-observer PRIVATE d3d11 dxgi ole32 oleaut32 uuid)
    target_link_libraries(ce-etl-frame-window PRIVATE tdh)
endif()
